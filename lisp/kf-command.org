(defun kf:vterm-opend ()
  (seq-find (lambda (b) (string= "*vterm*" (buffer-name b)))
    	    (buffer-list)))

(defun kf:vterm-cd (dir)
  (interactive)
  (with-current-buffer "*vterm*"
    (vterm-send-string (concat "cd " dir))
    (vterm-send-return)))

(defun kf:vterm-visit (dir)
  (interactive)
  (kf:vterm-cd dir)
  (switch-to-buffer-other-window "*vterm*"))

(defun kf:vterm-quit-wait ()
  "vterm に C-d を送信し、履歴が書き込まれてプロセスが終了するまで同期的に待機します。"
  (interactive)
  (let ((buf (kf:vterm-opend)))
    (when buf
      (let ((proc (get-buffer-process buf)))
        (when (and proc (process-live-p proc))
          ;; 1. vterm バッファを通じて bash に明確に C-d を送り、正規の終了（履歴書き込み）を促す
          (with-current-buffer buf
            (vterm-send-C-d))
          ;; 2. bash が終了処理（.bash_history への書き込み）を完遂して
          ;;    プロセスが完全に消滅するまで、最大2秒間（0.05秒刻みで）同期的に待機する
          (let ((counter 0))
            (while (and (process-live-p proc) (< counter 40))
              (sit-for 0.05)
              (setq counter (1+ counter))))
          ;; 3. プロセスが安全に消滅したことを確認してからバッファをクリーンアップする
          (unless (process-live-p proc)
            (kill-buffer buf)))))))

;; GUID: kf770b2c-4d3e-5f6a-8b9c-0d1e2f3a4b5c
;; ========================================================================
;; 外部変数による柔軟な除外パターン管理と、自動ノイズフィルター付き高速ファイル検索
;; ========================================================================

(defcustom kf:locate-ignore-patterns
  '("/\\.git/"
    "/\\.cache/"
    "/node_modules/"
    "~\$"                  ; バックアップファイル（末尾にチルダ）
    "^/tmp/"              ; テンポラリ領域
    )
  "`kf:locate' でファイル検索を行う際、検索結果から完全に除外（パージ）する正規表現のリスト。"
  :type '(repeat string)
  :group 'kf-commands)

(defun kf:locate (search-regexp)
  "locate コマンドをバックエンドに使い、ベースネームに対して正規表現検索を行います。
結果から `kf:locate-ignore-patterns' にマッチするパス、および『ディレクトリ自体』を完全に除外します。
入力例： kf-command\\.org\$"
  (interactive "sLocate (Regexp for Basename): ")
  (if (string-empty-p search-regexp)
      (user-error "検索クエリが空です")
    (let ((default-directory "/")
          (output-buffer (generate-new-buffer " kf-locate-output"))
          (filtered-lines nil)
          (exit-code nil))
      (unwind-protect
          (progn
            ;; -b (basename) と -r (regexp) を指定して locate を実行
            (setq exit-code (call-process "locate" nil output-buffer nil "-b" "-r" search-regexp))
            (if (not (eq exit-code 0))
                (message "locateコマンドがエラーを返しました（終了コード: %s）" exit-code)
              (with-current-buffer output-buffer
                (goto-char (point-min))
                (while (not (eobp))
                  (let ((line (buffer-substring-no-properties (line-beginning-position) (line-end-position))))
                    (unless (string-empty-p line)
                      ;; 【洗練されたリスト駆動フィルター】
                      ;; 外部変数に登録されたパターンのいずれか一つでもマッチするか、
                      ;; または対象がディレクトリである場合は即座にスキップします
                      (unless (or (seq-some (lambda (pattern) (string-match-p pattern line))
                                            kf:locate-ignore-patterns)
                                  (file-directory-p line))
                        (push line filtered-lines))))
                  (forward-line 1)))

              ;; フィルタリングされたクリーンな結果をユーザーに提示
              (if (null filtered-lines)
                  (message "マッチする有効なファイルは見つかりませんでした: %s" search-regexp)
                (let ((chosen-file (completing-read "Select file: " (nreverse filtered-lines) nil t)))
                  (when (and chosen-file (file-exists-p chosen-file))
                    (find-file-other-window chosen-file))))))

        ;; 一時バッファを確実にクリーンアップ
        (when (buffer-live-p output-buffer)
          (kill-buffer output-buffer))))))

#+begin_src elisp :tangle yes
;; GUID: kf660d5e-6a7b-8c9d-0e1f-2a3b4c5d6e7f
;; ========================================================================
;; 自動外枠パージ機能を搭載したマークアップ用インサート・置換コマンド
;; ========================================================================
(defvar kf:last-begin nil "最後に使用した開始マークアップ文字列")
(defvar kf:last-end nil "最後に使用した終了マークアップ文字列")

(defun kf:insert-begin-end (begin end)
  "選択リージョンの前後に BEGIN と END を挿入します。
もしリージョンの直前・直後が既に別のブロック（#+begin_... 等）で囲まれている場合は、
古いマークアップを自動的にパージ（除去）して上書き置換を行います。"
  (interactive
   (list (read-string "Begin: " kf:last-begin)
         (read-string "End: " kf:last-end)))
  (setq kf:last-begin begin
        kf:last-end end)

  (if (not (use-region-p))
      (user-error "リージョンが選択されていません")
    (let* ((start (region-beginning))
           (e (region-end))
           (has-outer-block nil)
           (outer-beg-pos nil)
           (outer-end-pos nil))

      ;; --- 鉄壁の既存マークアップ（外枠）事前検知ロジック ---
      (save-excursion
        ;; 1. リージョンの直前に "#+begin_" で始まる行があるかチェック
        (goto-char start)
        (forward-line -1)
        (let ((prev-line (buffer-substring-no-properties (line-beginning-position) (line-end-position))))
          (when (string-match-p "^\\s-*#\\+begin_" prev-line)
            (setq outer-beg-pos (cons (line-beginning-position) (1+ (line-end-position))))))

        ;; 2. リージョンの直後に "#+end_" で始まる行があるかチェック
        (goto-char e)
        ;; リージョン末尾が改行コード自体を掴んでいる場合の補正
        (when (and (= (char-before) ?\n) (< start (point))) (forward-char -1))
        (forward-line 1)
        (let ((next-line (buffer-substring-no-properties (line-beginning-position) (line-end-position))))
          (when (string-match-p "^\\s-*#\\+end_" next-line)
            (setq outer-end-pos (cons (line-beginning-position) (if (eobp) (line-end-position) (1+ (line-end-position))))))))

      ;; 上下の両方にマーカーが存在していれば、パージ（置換）モードを確定
      (setq has-outer-block (and outer-beg-pos outer-end-pos))

      ;; --- 本番のテキスト変形処理 ---
      (save-excursion
        (if has-outer-block
            ;; 【パターンA】上下の古いマークアップを消し去って、ダイレクトに上書き（添付3の挙動）
            (let ((content (buffer-substring-no-properties start e)))
              ;; 終了タグ行を削除
              (delete-region (car outer-end-pos) (cdr outer-end-pos))
              ;; 開始タグ行からコンテンツ末尾までを一旦すべて削除して綺麗にする
              (delete-region (car outer-beg-pos) e)
              ;; 新しい定義で再配置
              (goto-char (car outer-beg-pos))
              (insert begin "\n" content)
              ;; コンテンツ末尾が改行で終わっていない場合の安全な改行補正
              (if (= (char-before) ?\n) (insert end "\n") (insert "\n" end "\n"))
              (message "既存のブロック構造を %s に昇格・置換しました。" begin))

          ;; 【パターンB】周囲に何もないため、通常の新規挿入を行う（添付1から2への安全な移行）
          (goto-char e)
          (if (= (char-before) ?\n)
              (insert end "\n")
            (insert "\n" end "\n"))
          (goto-char start)
          (insert begin "\n")
          (message "新しく %s ブロックを適用しました。" begin))))))

(defun kf:extract-uppercase-words ()
  "現在開いているバッファから、2文字以上のすべて大文字の単語を重複なく抽出し、
別バッファにソートして出力します。"
  (interactive) ; これがあることで M-x からコマンドとして呼び出せるようになります
  (let ((words '())
        (case-fold-search nil) ; 大文字小文字を厳密に区別する
        (current-buf-name (buffer-name)))
    (save-excursion
      (goto-char (point-min))
      ;; 2文字以上の大文字（およびアンダースコア）のみを検索
      (while (re-search-forward "\\_<[A-Z_]\\{2,\\}\\_>" nil t)
        (add-to-list 'words (match-string-no-properties 0))))

    ;; 抽出結果を入れるバッファの作成と出力
    (if words
        (let ((output-buf (get-buffer-create "*Extracted Words*")))
          (with-current-buffer output-buf
            (erase-buffer)
            ;; 読みやすさのためにヘッダー（コメント）を挿入
            (insert (format ";; Extracted from %s\n" current-buf-name))
            (insert (mapconcat 'identity (sort words 'string<) "\n"))
            (insert "\n"))
          (pop-to-buffer output-buf)
          (message "すべて大文字の単語を %d 件抽出しました。" (length words)))

* markdown クリップボードをorgモード形式に変換してペースト

#+begin_src elisp :tangle yes
;; GUID: kf990e5f-9e2c-41f6-b8c3-a3d2f5b6c8fd
;; ========================================================================
;; ラインパーサー方式による文字化け・エラー皆無のMarkdown→Org変換ペースト
;; ========================================================================

(defun kf:paste-markdown-as-org ()
  "クリップボードのMarkdownを安全に取得し、行単位のクリーンな置換で確実にOrg文法に変換して挿入します。"
  (interactive)
  (let* ((cmd (cond
               ((executable-find "wl-paste") "wl-paste -t text/plain")
               ((executable-find "xclip") "xclip -selection clipboard -o -t UTF8_STRING")
               (t nil)))
         (text (when cmd
                 (let ((coding-system-for-read 'utf-8))
                   (shell-command-to-string cmd)))))

    (if (and text (not (string-empty-p text)))
        (with-temp-buffer
          ;; バッファをマルチバイトに固定し、文字列を安全に展開
          (set-buffer-multibyte t)
          (insert text)

          ;; 1. 【見出し変換】安全に行頭の # 列を検出し、タイトル文字列（.*）を完全に保持して * に置換
          (goto-char (point-min))
          (while (re-search-forward "^\\(#+\\) +.*\$" nil t)
            (let* ((hashes (match-string 1))
                   (title (match-string 2))
                   (stars (make-string (length hashes) ?*)))
              (replace-match (concat stars " " title) t t)))

          ;; 2. 【コードブロック変換】``` だけで構成される行を個別に置換（マルチライン正規表現のバグを回避）
          ;; 行頭の ```lang を検知して #+begin_src lang に変換します
          (goto-char (point-min))
          (while (re-search-forward "^```\\([a-zA-Z0-9_-]+\\)$" nil t)
            (replace-match "#+begin_src \\1" t))
          ;; 閉じの ``` または言語指定のない ``` 行を #+end_src / #+begin_example に安全トグル置換
          (goto-char (point-min))
          (while (re-search-forward "^```\$" nil t)
            (replace-match "#+end_src" t))

          ;; 3. 【インラインコード変換】`code` -> ~code~
          (goto-char (point-min))
          (while (re-search-forward "`\\([^`\n]+?\\)`" nil t)
            (replace-match "~\\1~"))

          ;; 4. 【太字変換】**text** -> *text*
          (goto-char (point-min))
          (while (re-search-forward "\\*\\*\\([^* \n][^*]*?[^* \n]\\)\\*\\*" nil t)
            (replace-match "*\\1*"))

          ;; 5. 【リンク変換】[text](URL) -> [[URL][text]]
          (goto-char (point-min))
          (while (re-search-forward "\\[\\([^]]+\\)\\](\\([^)]+\\))" nil t)
            (replace-match "[[\\2][\\1]]" t t))

          ;; 6. カレントバッファへの確実な挿入
          (let ((converted (buffer-string)))
            (with-current-buffer (car (buffer-list))
              (insert converted)
              (message "MarkdownからOrg文法への確実な変換ペーストが完了しました。"))))
      (message "エラー：クリップボードからUTF-8テキストデータを取得できませんでした。"))))

* org-block内のコンテンツリージョンを選択する
#+begin_src elisp :tangle yes
;; GUID: kf550c4e-5d6a-7b8c-9d0e-1f2a3b4c5d6f
;; ========================================================================
;; パース状態に依存せずブロックの内側を100%確実にリージョン選択するコマンド（修正版）
;; ========================================================================
(defun kf:select-org-block-content ()
  "現在のカーソルが属する Org ブロックの内側のコンテンツ範囲を、パース状態に関わらず確実に選択します。"
  (interactive)
  (if (not (derived-mode-p 'org-mode))
      (user-error "このコマンドは Org-mode でのみ利用可能です")
    (let* ((element (org-element-at-point))
           (type (org-element-type element)))
      ;; 対象が各種ブロック要素である場合のみ実行
      (if (memq type '(src-block example-block quote-block special-block center-block comment-block verse-block))
          (let ((element-beg (org-element-property :begin element))
                (element-end (org-element-property :end element))
                (beg nil)
                (end nil))
            (save-excursion
              ;; 1. ブロックの開始位置から、最初の #+begin_ 行の「次の行の先頭」をコンテンツ開始位置とする
              (goto-char element-beg)
              (while (and (< (point) element-end)
                          (not (looking-at "^\\s-*#\\+begin_")))
                (forward-line 1))
              (forward-line 1)
              (setq beg (point))

              ;; 2. ブロックの終了位置から、直前の #+end_ 行の「行頭（＝中身の改行終端）」をコンテンツ終了位置とする
              (goto-char element-end)
              ;; ブロック末尾の空白や改行を安全にスキップして #+end_ を探す
              (while (and (> (point) beg)
                          (not (looking-at "^\\s-*#\\+end_")))
                (forward-line -1))
              (setq end (point)))

            ;; 3. 計算された確定ポイントで確実にリージョンを選択
            (if (and beg end (< beg end))
                (progn
                  (goto-char beg)
                  (push-mark end t t)
                  (activate-mark)
                  (message "ブロック内のコンテンツ（%d〜%d）を確実に選択しました。" beg end))
              (user-error "ブロックの中身が空、または範囲を特定できませんでした")))
        (user-error "カーソルが Org ブロックの内部にありません")))))

* AI回答をソースブロック化
#+begin_src elisp :tangle yes
;; GUID: kf0b2c3d-4e5f-6a7b-8c9d-0e1f2a3b4c5d
;; ========================================================================
;; 【文字単位完全固定】1文字もコードを失わない定型コード自動Srcブロック化コマンド
;; ========================================================================
(defun kf:convert-plain-code-to-src-block ()
  "カーソル位置のブロックを全自動で解析し、コードの末尾のカッコを完全に維持して #+begin_src に構造化します。"
  (interactive)
  (if (not (derived-mode-p 'org-mode))
      (user-error "このコマンドは Org-mode のバッファでのみ実行可能です")
    (let ((case-fold-search nil)
          (end-marker "^コードは注意してご使用ください。")
          (beg-pos nil)
          (end-pos nil)
          (lang-name nil)
          (content ""))

      (save-excursion
        (beginning-of-line)
        ;; 1. カーソル位置から「下方向」へ、一番近い定型文（終了マーカー）を探す
        (if (re-search-forward end-marker nil t)
            (progn
              ;; 終了マーカーの「行末（改行含む全体）」を削除対象の終端（end-pos）とする
              (setq end-pos (match-end 0))

              ;; 🚨 【バグ対策】コードの本当の末尾（注意文の直前の行の末尾）の位置を厳密に保持
              (goto-char (match-beginning 0))
              (let ((code-end-point (point)))

                ;; 2. そこから「上方向」へ1行ずつ遡り、本物の言語名（アルファベット単独行）を探す
                (let ((searching t))
                  (while (and searching (not (bobp)))
                    (forward-line -1)
                    (let ((cur-line (string-trim (buffer-substring-no-properties (line-beginning-position) (line-end-position)))))
                      ;; コメントや記号を含まない、完全にアルファベット・数字・ハイフン等の「1単語のみの行」
                      (when (and (not (string-empty-p cur-line))
                                 (string-match-p "^[a-zA-Z0-9_-]+$" cur-line))
                        (setq lang-name cur-line)
                        ;; 言語名行の「行頭」を削除対象の先頭（beg-pos）とする
                        (setq beg-pos (line-beginning-position))

                        ;; 🚨 【バグ対策】言語名の「次の行の行頭」から、注意文の「直前の行の末尾」までを
                        ;; 文字数ベースで完璧に安全退避（content）。行移動による誤差を完全に根絶。
                        (save-excursion
                          (goto-char (line-end-position))
                          (forward-char 1)
                          (setq content (buffer-substring-no-properties (point) code-end-point)))
                        (setq searching nil)))))))
          (user-error "下方に終了マーカー『コードは注意してご使用ください。』が見つかりませんでした")))

      ;; --- 文字ポイントが完全に一致した場合のみ一括置換を実行 ---
      (if (and beg-pos end-pos lang-name)
          (save-excursion
            ;; 検出した範囲（言語名から定型文全体まで）を一括パージ
            (delete-region beg-pos end-pos)
            (goto-char beg-pos)
            ;; 退避させておいた純粋なコンテンツを一切損なわずにソースブロックとして再配置
            (insert (format "#+begin_src %s\n" (downcase lang-name)))
            (insert content)
            ;; コンテンツ末尾に必ず改行を維持して美しく閉じる
            (unless (= (char-before) ?\n)
              (insert "\n"))
            (insert "#+end_src\n")
            (message "コード末尾のカッコを完全に維持し、#+begin_src %s へ構造化しました！" lang-name))
        (user-error "上方の言語名（開始行）の自動検知に失敗しました。ファイル構造保護のため中断します")))))
