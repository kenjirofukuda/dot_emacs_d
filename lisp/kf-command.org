(defun kf:vterm-opend ()
  (seq-find (lambda (b) (string= "*vterm*" (buffer-name b)))
    	    (buffer-list)))

(defun kf:vterm-cd (dir)
  (interactive)
  (with-current-buffer "*vterm*"
    (vterm-send-string (concat "cd " dir))
    (vterm-send-return)))

(defun kf:vterm-visit (dir)
  (interactive)
  (kf:vterm-cd dir)
  (switch-to-buffer-other-window "*vterm*"))

(defun kf:vterm-quit-wait ()
  "vterm に C-d を送信し、履歴が書き込まれてプロセスが終了するまで同期的に待機します。"
  (interactive)
  (let ((buf (kf:vterm-opend)))
    (when buf
      (let ((proc (get-buffer-process buf)))
        (when (and proc (process-live-p proc))
          ;; 1. vterm バッファを通じて bash に明確に C-d を送り、正規の終了（履歴書き込み）を促す
          (with-current-buffer buf
            (vterm-send-C-d))
          ;; 2. bash が終了処理（.bash_history への書き込み）を完遂して
          ;;    プロセスが完全に消滅するまで、最大2秒間（0.05秒刻みで）同期的に待機する
          (let ((counter 0))
            (while (and (process-live-p proc) (< counter 40))
              (sit-for 0.05)
              (setq counter (1+ counter))))
          ;; 3. プロセスが安全に消滅したことを確認してからバッファをクリーンアップする
          (unless (process-live-p proc)
            (kill-buffer buf)))))))

(defun kf:open-file-thing ()
  "カーソル行のファイルを開く"
  (interactive)
  (let* ((url-part (kf:get-org-url-part-in-caption))
  	 (filename (or (thing-at-point 'filename t)
  		       (and url-part
  			    (string-match-p "^file:///" url-part))))
  	 (url (or (thing-at-point 'url t)
  		  url-part)))
    (when url
      (cond ((string-match-p "^file:///" url)
             (eww-browse-url url)
             (message "eww: %s" url))
            (t (message "chrome: %s" url)
               (browse-url-chrome url))))
    (when (and filename (file-exists-p filename))
      (cond ((string-match-p "\\.wav$" filename)
             (play-sound-file filename))
            (t
             (find-file-other-window filename))))))

(defun kf:get-org-url-part-in-caption ()
    ;;; (interactive)
  (defun point-in-range-p (range)
    (and (>= (current-column) (car range)) (< (current-column) (cdr range))))
  (let* ((thing (thing-at-point 'line))
  	 (matched (string-match  "\\[\\[\\(.+\\)\\]\\[\\(.+\\)\\]\\]" thing)))
    (if matched
  	(let ((url-range (cons (match-beginning 1) (match-end 1)) )
  	      (caption-range (cons (match-beginning 2) (match-end 2))))
  	  (if (point-in-range-p caption-range)
  	      ;;(list url-range caption-range)
  	      (substring thing (car url-range) (cdr url-range)))))))

(defun kf:locate (search-string)
  "仮想コンソールを開くことなく、locateコマンドを非同期に実行してファイルを高速検索します。
結果はミニバッファで一行ずつ選択（completing-read）でき、選択したファイルを即座に開きます。"
  (interactive "sLocate file: ")
  (if (string-empty-p search-string)
      (user-error "検索クエリが空です")
    (let ((default-directory "/")
	  (output-buffer (generate-new-buffer " kf-locate-output"))
	  (exit-code nil))
      (unwind-protect
	  (progn
	    ;; 同期または非同期でlocateコマンドの出力を一時バッファに格納
	    ;; 高速なインデックス検索のため、通常のファイル走査より著しく高速です
	    (setq exit-code (call-process "locate" nil output-buffer nil search-string))
	    (if (not (eq exit-code 0))
		(message "locateコマンドがエラーを返しました（終了コード: %s）" exit-code)
	      (with-current-buffer output-buffer
		(let ((lines nil))
		  (goto-char (point-min))
		  (while (not (eobp))
		    (let ((line (buffer-substring-no-properties (line-beginning-position) (line-end-position))))
		      (unless (string-empty-p line)
			(push line lines)))
		    (forward-line 1))
		  (if (null lines)
		      (message "マッチするファイルは見つかりませんでした: %s" search-string)
		    ;; 補完インターフェース（Vertico, Selectrum, 又は標準のcompleting-read）で選択
		    (let ((chosen-file (completing-read "Select file: " (nreverse lines) nil t)))
		      (when (and chosen-file (file-exists-p chosen-file))
			(find-file chosen-file))))))))
	;; 一時バッファを確実にクリーンアップ
	(when (buffer-live-p output-buffer)
	  (kill-buffer output-buffer))))))
(defun kf:updatedb ()
  "仮想コンソールを開くことなく、locateコマンドのデータベース(updatedb)を非同期プロセスで更新します。

Debian環境でpolkitが設定されている場合は、グラフィカル、またはミニバッファでのパスワード要求が発生します。"
  (interactive)
  (message "locate データベース(updatedb)を非同期で更新中...")

(defun kf:extract-uppercase-words ()
  "現在開いているバッファから、2文字以上のすべて大文字の単語を重複なく抽出し、
別バッファにソートして出力します。"
  (interactive) ; これがあることで M-x からコマンドとして呼び出せるようになります
  (let ((words '())
        (case-fold-search nil) ; 大文字小文字を厳密に区別する
        (current-buf-name (buffer-name)))
    (save-excursion
      (goto-char (point-min))
      ;; 2文字以上の大文字（およびアンダースコア）のみを検索
      (while (re-search-forward "\\_<[A-Z_]\\{2,\\}\\_>" nil t)
        (add-to-list 'words (match-string-no-properties 0))))

    ;; 抽出結果を入れるバッファの作成と出力
    (if words
        (let ((output-buf (get-buffer-create "*Extracted Words*")))
          (with-current-buffer output-buf
            (erase-buffer)
            ;; 読みやすさのためにヘッダー（コメント）を挿入
            (insert (format ";; Extracted from %s\n" current-buf-name))
            (insert (mapconcat 'identity (sort words 'string<) "\n"))
            (insert "\n"))
          (pop-to-buffer output-buf)
          (message "すべて大文字の単語を %d 件抽出しました。" (length words)))
      (message "すべて大文字の単語は見つかりませんでした。"))))

* markdown クリップボードをorgモード形式に変換してペースト

#+begin_src elisp :tangle yes
;; GUID: kf990e5f-9e2c-41f6-b8c3-a3d2f5b6c8fd
;; ========================================================================
;; クリップボードのマークダウン（見出し対応版）をOrg-mode文法に自動変換してペースト
;; ========================================================================

(defun kf:paste-markdown-as-org ()
  "クリップボード内のMarkdownテキストを取得し、見出しを含む文法をOrg-modeに変換して挿入します。"
  (interactive)
  (let ((text (gui-get-selection 'CLIPBOARD 'STRING)))
    (if (and text (not (string-empty-p text)))
        (with-temp-buffer
          ;; 1. 一時バッファにクリップボードのテキストを挿入
          (insert text)
          (goto-char (point-min))

          ;; 2. マークダウンの見出し (^#+ ) を Orgの見出し (* ) に動的変換
          ;; 行頭の # の数をカウントし、同じ数の * に置換します
          (while (re-search-forward "^\\(#+\\) +.*\$" nil t)
            (let* ((hashes (match-string 1))
                   (title-text (match-string 2))
                   (stars (make-string (length hashes) ?*)))
              (replace-match (concat stars title-text) t t)))
          (goto-char (point-min))

          ;; 3. マークダウンの太字 (**text**) を Orgの強調 (*text*) に変換
          (while (re-search-forward "\\*\\*\\([^* \n][^*]*?[^* \n]\\)\\*\\*" nil t)
            (replace-match "*\\1*"))
          (goto-char (point-min))

          ;; 4. マークダウンのインラインコード (`code`) を Orgのコード (~code~) に変換
          (while (re-search-forward "`\\([^`\n]+?\\)`" nil t)
            (replace-match "~\\1~"))
          (goto-char (point-min))

          ;; 5. マークダウンのコードブロック (```lang ... ```) を Org-babel (#+begin_src) に変換
          (while (re-search-forward "^```\\([a-zA-Z0-9_-]*\\)\n\\(\\(?:\n\Vert{}.\\)*?\\)```" nil t)
            (let ((lang (match-string 1))
                  (body (match-string 2)))
              (if (string-empty-p lang)
                  (replace-match (format "#+begin_example\n%s#+end_example" body) t t)
                (replace-match (format "#+begin_src %s\n%s#+end_src" lang body) t t))))

          ;; 6. 変換後のテキストを現在のOrgバッファのカーソル位置へ挿入
          (let ((converted-text (buffer-string)))
            (with-current-buffer (current-buffer)
              (insert converted-text)
              (message "Markdownの見出し・コードをOrg文法へ変換してペーストしました。"))))
      (message "エラー：クリップボードが空か、有効なテキストデータがありません。"))))

#+end_src
